   111→type TrendPoint = AnalyticsSummary["trend"][number];

   113→type TrendDirection = "up" | "down" | "flat";
   114→type TrendInfo = { pct: number; favorable: boolean; direction: TrendDirection } | null;
   115→
   116→// Compute the % change of a metric comparing the last 7 days vs the
   117→// previous 7 days from the 14-day trend. Returns null when there's no
   118→// prior data to compare against (prev7 === 0). `favorable` is determined
   119→// by the caller-supplied direction ("up" or "down") that counts as good.
   120→function computeTrendPct(
   121→  trend: AnalyticsSummary["trend"],
   122→  accessor: (d: TrendPoint) => number,
   123→  favorableDirection: "up" | "down",
   124→): TrendInfo {
   125→  if (trend.length < 14) return null;
   126→  const last7 = trend.slice(-7).reduce((s, d) => s + accessor(d), 0);
   127→  const prev7 = trend.slice(-14, -7).reduce((s, d) => s + accessor(d), 0);
   128→  if (prev7 === 0) return null;
   129→  const pct = ((last7 - prev7) / Math.abs(prev7)) * 100;
   130→  const direction: TrendDirection =
   131→    pct > 0.5 ? "up" : pct < -0.5 ? "down" : "flat";
   132→  const favorable =
   133→    direction === "flat" ? true : direction === favorableDirection;
   134→  return { pct, direction, favorable };
   135→}

   137→// Build the cumulative open backlog sparkline (filed − resolved per day,
   138→// accumulated and clamped at 0 so it never implies negative complaints).
   139→function buildOpenSparkline(trend: AnalyticsSummary["trend"]): number[] {
   140→  let cumulative = 0;
   141→  const out: number[] = [];
   142→  for (const d of trend) {
   143→    cumulative += d.filed - d.resolved;
   144→    if (cumulative < 0) cumulative = 0;
   145→    out.push(cumulative);
   146→  }
   147→  return out;
   148→}

   150→// Type alias for the translate function so helpers can use it without

   153→
   154→// Format "X min ago" relative to a caller-supplied `now` (so the component
   155→// can re-evaluate the relative time on a 60s tick). Mirrors the logic in
   156→// lib/format.ts formatRelative but accepts an explicit now.
   157→function relativeAgo(
   158→  then: number,
   159→  now: number,
   160→  locale: Locale,
   161→  t: TFunc,
   162→): string {
   163→  const diff = Math.max(0, now - then);
   164→  const minutes = Math.floor(diff / 60000);
   165→  const hours = Math.floor(diff / 3600000);
   166→  const days = Math.floor(diff / 86400000);
   167→  if (minutes < 1) return locale === "en" ? "just now" : t("common.justNow");
   168→  if (minutes < 60) return t("common.minutesAgo", { n: minutes });
   169→  if (hours < 24) return t("common.hoursAgo", { n: hours });
   170→  if (days < 30) return t("common.daysAgo", { n: days });
   171→  return new Date(then).toLocaleDateString(locale, {
   172→    year: "numeric",
   173→    month: "short",
   174→    day: "numeric",
   175→  });
   176→}

   555→// ---------------------------------------------------------------------------
   556→type KpiTone = "default" | "open" | "resolved" | "sla";
   557→
   558→function KpiCard({
   559→  icon: Icon,
   560→  label,
   561→  value,
   562→  suffix,
   563→  tone,
   564→  sparkData,
   565→  sparkColor,
   566→  trend,
   567→  t,
   568→  index,
   569→}: {
   570→  icon: React.ComponentType<{ className?: string }>;
   571→  label: string;
   572→  value: number;
   573→  suffix?: string;
   574→  tone: KpiTone;
   575→  sparkData: number[];
   576→  sparkColor: string;
   577→  trend: TrendInfo;
   578→  t: ReturnType<typeof useTranslations>["t"];
   579→  index: number;
   580→}) {
   581→  const toneClasses: Record<KpiTone, string> = {
   582→    default: "border-border bg-card",
   583→    open: "border-amber-200 bg-amber-50/50 dark:border-amber-900/40 dark:bg-amber-950/20",
   584→    resolved: "border-emerald-200 bg-emerald-50/50 dark:border-emerald-900/40 dark:bg-emerald-950/20",
   585→    sla: "border-primary/30 bg-primary/5",
   586→  };
   587→  const iconClasses: Record<KpiTone, string> = {
   588→    default: "bg-primary/10 text-primary",
   589→    open: "bg-amber-500/15 text-amber-600 dark:text-amber-400",
   590→    resolved: "bg-emerald-500/15 text-emerald-600 dark:text-emerald-400",
   591→    sla: "bg-primary/15 text-primary",
   592→  };
   593→
   594→  return (
   595→    <motion.div
   596→      initial={{ opacity: 0, y: 8 }}
   597→      animate={{ opacity: 1, y: 0 }}
   598→      transition={{ duration: 0.3, delay: Math.min(index * 0.05, 0.25), ease: "easeOut" }}
   599→    >
   600→      <Card className={cn("h-full py-4", toneClasses[tone])}>
   601→        <CardContent className="space-y-3">
   602→          <span
   603→            className={cn(
   604→              "flex size-8 items-center justify-center rounded-lg",
   605→              iconClasses[tone],
   606→            )}
   607→          >
   608→            <Icon className="size-4" aria-hidden />
   609→          </span>
   610→          <div>
   611→            <p className="text-2xl font-semibold leading-none tabular-nums text-foreground">
   612→              {value}
   613→              {suffix ? (
   614→                <span className="ml-0.5 text-sm font-normal text-muted-foreground">{suffix}</span>
   615→              ) : null}
   616→            </p>
   617→            <p className="mt-1 text-xs text-muted-foreground">{label}</p>
   618→          </div>
   619→          <div className="flex items-end justify-between gap-2">
   620→            {sparkData.length > 0 ? (
   621→              <Sparkline data={sparkData} color={sparkColor} />
   622→            ) : (
   623→              <span className="h-10" aria-hidden />
   624→            )}
   625→            <TrendBadge trend={trend} t={t} />
   626→          </div>
   627→        </CardContent>
   628→      </Card>
   629→    </motion.div>
   630→  );
   631→}

   633→// ---------------------------------------------------------------------------
   634→// Mini line chart — no axes, no grid, no tooltip. Pure trend shape.
   635→// ---------------------------------------------------------------------------
   636→function Sparkline({ data, color }: { data: number[]; color: string }) {
   637→  if (data.length < 2) return null;
   638→  const chartData = data.map((v, i) => ({ i, v }));
   639→  return (
   640→    <div
   641→      style={{ height: SPARK_HEIGHT, width: 80 }}
   642→      role="img"
   643→      aria-label="14-day trend"
   644→    >
   645→      <ResponsiveContainer width="100%" height="100%">
   646→        <LineChart data={chartData} margin={{ top: 2, right: 2, bottom: 2, left: 2 }}>
   647→          <Line
   648→            type="monotone"
   649→            dataKey="v"
   650→            stroke={color}
   651→            strokeWidth={1.5}
   652→            dot={false}
   653→            isAnimationActive={false}
   654→          />
   655→        </LineChart>
   656→      </ResponsiveContainer>
   657→    </div>
   658→  );
   659→}

   661→// ---------------------------------------------------------------------------
   662→// Trend badge — pill showing % change vs last week. Green when favorable,
   663→// amber when unfavorable. Direction icon reflects the actual change.
   664→// ---------------------------------------------------------------------------
   665→function TrendBadge({
   666→  trend,
   667→  t,
   668→}: {
   669→  trend: TrendInfo;
   670→  t: ReturnType<typeof useTranslations>["t"];
   671→}) {
   672→  if (!trend) {
   673→    return (
   674→      <Badge
   675→        variant="outline"
   676→        className="border-border bg-muted/40 text-muted-foreground"
   677→      >
   678→        <span className="tabular-nums">{t("admin.noTrend")}</span>
   679→      </Badge>
   680→    );
   681→  }
   682→  const DirectionIcon = trend.direction === "up" ? TrendingUp : trend.direction === "down" ? TrendingDown : null;
   683→  const toneClass = trend.favorable
   684→    ? "border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-900 dark:bg-emerald-950/40 dark:text-emerald-300"
   685→    : "border-amber-200 bg-amber-50 text-amber-700 dark:border-amber-900 dark:bg-amber-950/40 dark:text-amber-300";
   686→  const pctLabel = trend.direction === "up" ? t("admin.trendUp", { pct: Math.round(Math.abs(trend.pct)) }) : t("admin.trendDown", { pct: Math.round(Math.abs(trend.pct)) });
   687→  return (
   688→    <Badge variant="outline" className={cn("gap-0.5 px-1.5", toneClass)} title={t("admin.vsLastWeek")}>
   689→      {DirectionIcon ? <DirectionIcon className="size-3" aria-hidden /> : null}
   690→      <span className="tabular-nums">{pctLabel}</span>
   691→    </Badge>
   692→  );
   693→}

   718→// ---------------------------------------------------------------------------
   719→// Collapsible wrapper for a chart's sibling text table. Defaults to closed.
   720→// The table's sr-only <TableCaption> stays present in the DOM for screen
   721→// readers even when collapsed — only the visible rows toggle.
   722→// ---------------------------------------------------------------------------
   723→function CollapsibleTable({
   724→  showLabel,
   725→  hideLabel,
   726→  children,
   727→}: {
   728→  showLabel: string;
   729→  hideLabel: string;
   730→  children: React.ReactNode;
   731→}) {
   732→  const [open, setOpen] = useState(false);
   733→  return (
   734→    <Collapsible open={open} onOpenChange={setOpen} className="space-y-2">
   735→      <CollapsibleTrigger asChild>
   736→        <Button
   737→          variant="ghost"
   738→          size="sm"
   739→          className="h-7 gap-1 px-2 text-xs font-medium text-muted-foreground hover:text-foreground"
   740→          aria-expanded={open}
   741→        >
   742→          {open ? (
   743→            <ChevronUp className="size-3.5" aria-hidden />
   744→          ) : (
   745→            <ChevronDown className="size-3.5" aria-hidden />
   746→          )}
   747→          {open ? hideLabel : showLabel}
   748→        </Button>
   749→      </CollapsibleTrigger>
   750→      <CollapsibleContent>{children}</CollapsibleContent>
   751→    </Collapsible>
   752→  );
   753→}
